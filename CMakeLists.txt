cmake_minimum_required(VERSION 3.16)
project(RAMGPTQuantBench LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/llama.cpp)

add_library(ramgpt_formats cpp/ramgpt-logit-probe/formats.cpp)
target_include_directories(ramgpt_formats PUBLIC cpp/ramgpt-logit-probe)

add_executable(ramgpt-tokenize cpp/ramgpt-logit-probe/tokenize.cpp)
target_link_libraries(ramgpt-tokenize PRIVATE ramgpt_formats llama)
add_executable(ramgpt-logit-probe cpp/ramgpt-logit-probe/probe.cpp)
target_link_libraries(ramgpt-logit-probe PRIVATE ramgpt_formats llama)
add_executable(ramgpt-compare-reference cpp/ramgpt-logit-probe/compare.cpp)
target_link_libraries(ramgpt-compare-reference PRIVATE ramgpt_formats)
add_library(ramgpt_metrics cpp/ramgpt-logit-probe/metrics.cpp)
target_include_directories(ramgpt_metrics PUBLIC cpp/ramgpt-logit-probe)
add_executable(ramgpt-quant-compare cpp/ramgpt-logit-probe/quant_compare.cpp)
target_link_libraries(ramgpt-quant-compare PRIVATE ramgpt_formats ramgpt_metrics llama)
add_executable(ramgpt-analyze-fidelity cpp/ramgpt-logit-probe/analyze_fidelity.cpp)
target_link_libraries(ramgpt-analyze-fidelity PRIVATE ramgpt_formats ramgpt_metrics)
add_executable(ramgpt-inspect-tokens cpp/ramgpt-logit-probe/inspect_tokens.cpp)
target_link_libraries(ramgpt-inspect-tokens PRIVATE ramgpt_formats llama)
add_executable(ramgpt-decode-rqseg cpp/ramgpt-logit-probe/decode_rqseg.cpp)
target_link_libraries(ramgpt-decode-rqseg PRIVATE ramgpt_formats llama)
add_executable(ramgpt-decode-token-ids cpp/ramgpt-logit-probe/decode_token_ids.cpp)
target_link_libraries(ramgpt-decode-token-ids PRIVATE llama)
add_executable(ramgpt-format-tests tests/format_tests.cpp)
target_link_libraries(ramgpt-format-tests PRIVATE ramgpt_formats ramgpt_metrics)

enable_testing()
add_test(NAME ramgpt-format-tests COMMAND ramgpt-format-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME rql-python-compat
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_rql_python_compat.py $<TARGET_FILE:ramgpt-format-tests> ${CMAKE_CURRENT_SOURCE_DIR})
